2677. Chunk Array

/**
 * @param {Array} arr
 * @param {number} size
 * @return {Array}
 */
var chunk = function (arr, size) {
    let result = [];

    if (arr.length === 0) {
        return result;
    }

    if (size > arr.length) {
        return [arr];
    }

    let tempSize = size;
    let subArr = [];
    for (let i = 0; i < arr.length; i++) {
        if (tempSize > 0) {
            subArr.push(arr[i]);
            tempSize--;
        } else {
            result.push([...subArr]);
            subArr.length = 0;
            tempSize = size;

            subArr.push(arr[i]);
            tempSize--;
        }
    }
    result.push(subArr);

    return result;
};



/*
T(n) = O(n^2)
S(n) = O(n)
*/

var chunk = function(arr, size) {
    const result = [];

    for (let i = 0; i < arr.length; i += size) {
        result.push(arr.slice(i, i + size));
    }

    return result;
};
