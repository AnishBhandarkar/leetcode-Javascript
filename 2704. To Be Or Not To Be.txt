2704. To Be Or Not To Be


/**
 * @param {string} val
 * @return {Object}
 */
var expect = function (val) {
    return {
        toBe: (data) => {
            if (data === val) {
                return true;
            }
            throw new Error("Not Equal");
        },
        notToBe: (data) => {
            if (data !== val) {
                return true;
            }
            throw new Error("Equal");
        },
    }
};

/**
 * expect(5).toBe(5); // true
 * expect(5).notToBe(5); // throws "Equal"
 */
